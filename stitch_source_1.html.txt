<!-- Interactive Client-side Script for Filtering and Search -->
<script>
    (function initDashboard() {
      const filterButtons = document.querySelectorAll('.filter-btn');
      const tableRows = document.querySelectorAll('#documentsTable tbody tr');
      const searchInput = document.getElementById('tableSearch');

      let currentFilter = 'all';
      let currentQuery = '';

      function updateTable() {
        tableRows.forEach(row => {
          const category = row.getAttribute('data-category');
          const text = row.innerText.toLowerCase();

          const matchesCategory = (currentFilter === 'all') || (category === currentFilter);
          const matchesQuery = currentQuery === '' || text.includes(currentQuery);

          if (matchesCategory && matchesQuery) {
            row.style.display = '';
          } else {
            row.style.display = 'none';
          }
        });
      }

      filterButtons.forEach(btn => {
        btn.addEventListener('click', () => {
          filterButtons.forEach(b => {
            b.classList.remove('bg-surface-container-lowest', 'text-primary', 'shadow-sm', 'font-semibold');
            b.classList.add('text-on-surface-variant');
          });
          btn.classList.add('bg-surface-container-lowest', 'text-primary', 'shadow-sm', 'font-semibold');
          btn.classList.remove('text-on-surface-variant');

          currentFilter = btn.getAttribute('data-status');
          updateTable();
        });
      });

<script>
  // Interactive UI Script for DocuFlow Bulk Ingestion
  document.addEventListener('DOMContentLoaded', () => {
    // Pill selection logic
    const schemaPills = document.querySelectorAll('.schema-pill');
    schemaPills.forEach(pill => {
      pill.addEventListener('click', () => {
        schemaPills.forEach(p => {
          p.classList.remove('bg-primary', 'text-on-primary', 'shadow-sm');
          p.classList.add('bg-surface-container-low', 'text-on-surface-variant');
        });
        pill.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
        pill.classList.add('bg-primary', 'text-on-primary', 'shadow-sm');
      });
    });

    // Dropzone click triggers hidden file input
    const dropZone = document.getElementById('drop-zone');
    const fileInput = document.getElementById('bulk-file-input');
    
    dropZone.addEventListener('click', (e) => {
      // Prevent triggering file dialog when clicking options checkboxes inside dropzone
      if (e.target.closest('input[type="checkbox"]') || e.target.closest('label')) return;
      fileInput.click();
    });

    // Drag-over styling
    dropZone.addEventListener('dragover', (e) => {
      e.preventDefault();
      dropZone.classList.add('bg-surface-container');
    });

    dropZone.addEventListener('dragleave', () => {
      dropZone.classList.remove('bg-surface-container');
    });

    dropZone.addEventListener('drop', (e) => {
      e.preventDefault();
      dropZone.classList.remove('bg-surface-container');
      // Visual mock interaction
      alert('Documents dropped successfully. Running SHA-256 pre-check...');
    });

    // Queue Row Actions (Removal & Override)
    const queueList = document.getElementById('staged-queue-list');
    
    function updateQueueSummary() {
      const rows = queueList.querySelectorAll('.file-queue-row');
      const badgeCount = document.getElementById('queue-badge-count');
      badgeCount.textContent = `${rows.length} Items`;

      let readyCount = 0;
      let warningCount = 0;
      let totalSizeMB = 0;

      rows.forEach(row => {
        const isValid = row.getAttribute('data-valid') === 'true';
        const size = parseFloat(row.getAttribute('data-size') || 0);
        if (isValid) {
          readyCount++;
          totalSizeMB += size;
        } else {
          warningCount++;
        }
      });

      const summaryText = document.getElementById('queue-summary-text');
      const startBtn = document.getElementById('start-process-btn');
      
      if (rows.length === 0) {
        summaryText.innerHTML = 'Queue is empty • Ready for upload';
        startBtn.classList.add('opacity-50', 'pointer-events-none');
        startBtn.innerHTML = '<span class="material-symbols-outlined text-lg">bolt</span><span>Start Processing (0 Documents)</span>';
      } else {
        startBtn.classList.remove('opacity-50', 'pointer-events-none');
        summaryText.innerHTML = `${readyCount} files ready to process (${totalSizeMB.toFixed(1)} MB) <span class="text-outline font-normal">•</span> <span class="${warningCount > 0 ? 'text-error' : 'text-outline'} font-medium">${warningCount} warnings</span>`;
        startBtn.innerHTML = `<span class="material-symbols-outlined text-lg">bolt</span><span>Start Processing (${readyCount} Documents)</span>`;
      }
    }

    queueList.addEventListener('click', (e) => {
      const removeBtn = e.target.closest('.remove-row-btn');
      if (removeBtn) {
        const row = removeBtn.closest('.file-queue-row');
        row.style.opacity = '0';
        row.style.transform = 'translateX(20px)';
        row.style.transition = 'all 0.2s ease-out';
        setTimeout(() => {
          row.remove();
          updateQueueSummary();
        }, 200);
      }

      const overrideBtn = e.target.closest('.override-btn');
      if (overrideBtn) {
        const row = overrideBtn.closest('.file-queue-row');
        row.setAttribute('data-valid', 'true');
        row.classList.remove('bg-surface-container-high/40');
        row.classList.add('hover:bg-surface-container-low/60');
        
        const statusContainer = row.querySelector('.w-full.md\\:w-3\\/12');
        if (statusContainer) {
          statusContainer.innerHTML = `
            <span class="px-space-xs py-0.5 rounded font-label-sm text-label-sm uppercase tracking-wider font-semibold bg-tertiary-container/15 text-tertiary flex items-center gap-1">
              <span class="material-symbols-outlined text-xs">check_circle</span> Overridden (Ready)
            </span>
          `;
        }
        overrideBtn.remove();
        updateQueueSummary();
      }
    });

    // Clear Queue
    document.getElementById('clear-queue-btn').addEventListener('click', () => {
      if (confirm('Are you sure you want to clear all staged documents?')) {
        queueList.innerHTML = '<div class="p-space-lg text-center font-body-md text-body-md text-outline">No documents in queue. Drag files into the box above to stage for processing.</div>';
        updateQueueSummary();
      }
    });

    // Start Processing Handler
    document.getElementById('start-process-btn').addEventListener('click', function() {
      this.innerHTML = '<span class="material-symbols-outlined animate-spin text-lg">progress_activity</span><span>Ingesting & Allocating Nodes...</span>';
      this.classList.add('bg-tertiary-container', 'pointer-events-none');
      setTimeout(() => {
        alert('Ingestion dispatched successfully! 3 documents queued for OCR parsing and confidence extraction.');
      }, 700);
    });
  });
